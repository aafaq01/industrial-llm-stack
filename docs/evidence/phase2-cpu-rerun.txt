## Design
For each concurrency 1, 4, 8:
- COLD x3: docker restart llamacpp, wait for healthy (counters back to 0), then one run.py call with --repeats 1. Each
  cold repeat starts with an empty prompt cache. Because they are separate run.py calls, all three raw files carry
  "repeat": 1; report.py orders them by start time and shows them as #1, #2, #3.
- Warming pass: restart, then one full run.py call with --smoke into warming/ (discarded: report.py skips smoke runs;
  the files are kept for traceability).
- WARM x3: one run.py call with --repeats 3, no restart (repeats r1, r2, r3).
Host sampler (scratchpad hostsampler.ps1, PowerShell Get-Counter) every ~10 s for the whole run:
  \Processor(_Total)\% Processor Time, \Processor Information(_Total)\% Processor Performance (effective clock as % of
  the 2.0 GHz base), Processor Frequency, \Thermal Zone Information(*)\Temperature (ACPI zone, K converted to C),
  % Passive Limit; top 5 processes every ~60 s.
  -> benchmark/results/raw/cpu-20261009/host-samples.csv and host-top-processes.txt (CRLF converted to LF).
Driver script (scratchpad rerun.sh, run in the background), full text:
#!/usr/bin/env bash
# Phase 2 CPU rerun: per concurrency, 3 COLD repeats (llamacpp restarted before each) then a discarded warming pass
# and 3 WARM repeats without restarts. Raw output goes outside the repo during the run (keeps git_commit clean) and
# is copied to benchmark/results/raw/<session>/ afterwards.
set -euo pipefail
cd /c/Users/<user>/Desktop/work/industrial-llm-platform
S=$1 OUT=$2
C=industrial-llm-platform-llamacpp-1
W=$(cygpath -w "$S")
rm -f "$S/stop"
powershell.exe -NoProfile -ExecutionPolicy Bypass -File "$W\hostsampler.ps1" -Out "$W\host-samples.csv" \
  -TopOut "$W\host-top-processes.txt" -Stop "$W\stop" &
SAMPLER=$!
ts() { date -u +%Y-%m-%dT%H:%M:%SZ; }
restart() {
  echo "restart llamacpp at $(ts)"
  docker restart "$C" >/dev/null
  for _ in $(seq 1 120); do
    [ "$(docker inspect -f '{{.State.Health.Status}}' $C)" = healthy ] && break; sleep 2
  done
  echo "healthy at $(ts); $(curl -fs http://127.0.0.1:8080/metrics | grep -E '^llamacpp:prompt_tokens(_cached)?_total' | tr '\n' ' ')"
}
run() {  # conc repeats out-dir notes [extra]
  benchmark/.venv/Scripts/python benchmark/loadgen/run.py --base-url http://127.0.0.1:8080/v1 \
    --model qwen2.5-1.5b-instruct-q4_k_m \
    --model-revision sha256:6a1a2eb6d15622bf3c96857206351ba97e1af16c30d7a74ee38970e434e9407e \
    --engine llama.cpp --engine-version b11459 --precision gguf-q4_k_m --label llamacpp-cpu-1.5b-q4km \
    --hardware "Windows 11 host, Docker Desktop WSL2, llama.cpp --threads 4 --parallel 2" \
    --concurrency "$1" --repeats "$2" --metrics-url http://127.0.0.1:8080/metrics \
    --server-args "llama-server --host 0.0.0.0 --port 8080 --metrics -m /models/qwen2.5-1.5b-instruct-q4_k_m.gguf --ctx-size 4096 --parallel 2 --threads 4" \
    --notes "$4" --out-dir "$3" "${@:5}"
}
echo "rerun start $(ts)"
for conc in 1 4 8; do
  for i in 1 2 3; do
    echo "== c$conc COLD $i/3"
    restart
    run "$conc" 1 "$OUT" "Phase 2 CPU rerun, COLD $i/3: llamacpp restarted right before this run (empty prompt cache)."
  done
  echo "== c$conc warming pass (discarded, marked smoke)"
  restart
  run "$conc" 1 "$OUT/warming" "Phase 2 CPU rerun: discarded warming pass before the WARM repeats." --smoke
  echo "== c$conc WARM x3 (no restarts)"
  run "$conc" 3 "$OUT" "Phase 2 CPU rerun, WARM: after a discarded full warming pass, no restart; prompt cache warm."
done
echo "rerun end $(ts)"
touch "$S/stop"; wait $SAMPLER || true
echo "sampler stopped $(ts)"
